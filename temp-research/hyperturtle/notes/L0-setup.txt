1. OS and Hardware Verification

# Verify Ubuntu 20.04.6 LTS is running
lsb_release -a

# Verify Simultaneous Multithreading (SMT) is disabled (Should output 'off')
cat /sys/devices/system/cpu/smt/control

# Check current CPU frequency (Should be pinned around ~2.1GHz)
cpupower frequency-info | grep "current CPU frequency"


2. Kernel and Boot Parameters

# Verify you are booted into the custom HyperTurtle kernel (Not a generic Ubuntu kernel)
uname -r

# Verify custom BPF configs were applied to this kernel
grep CONFIG_BPF /boot/config-$(uname -r)

# Check that 'idle=poll' and 'nopti' are present in the boot parameters
cat /proc/cmdline | grep -o -E 'idle=poll|nopti'


3. Virtualization and Network Modules

# Check KVM Pause Loop Exiting (PLE) gap (Should output '0')
cat /sys/module/kvm_intel/parameters/ple_gap

# Check vhost_net Zero Copy Tx (Should output '1')
cat /sys/module/vhost_net/parameters/experimental_zcopytx


4. libbpf Installation

# Check the dynamic linker cache to ensure libbpf is accessible
ldconfig -p | grep libbpf

# Verify the libbpf development headers are installed
ls -l /usr/include/bpf/bpf.h || ls -l /usr/local/include/bpf/bpf.h


5. Custom QEMU Build

# Verify the custom QEMU binary compiled successfully for the right target

# Verify QEMU is dynamically linked against your custom libbpf (CRITICAL)

#!/bin/bash
set -e

# 1. Define Paths
SYS_BPF_HEADER="/usr/include/bpf/bpf_helper_defs.h"
HT_BPF_DIR="/work/hyperturtle/hyperturtle-linux/tools/lib/bpf"
HT_BPF_HEADER="$HT_BPF_DIR/bpf_helper_defs.h"
HELPER_DEF="static long (*bpf_probe_read_hyperupcall)(void *dst, __u32 size, const void *unsafe_ptr) = (void *) 180;"
QEMU_FILE="/work/hyperturtle/hyperturtle-qemu/accel/kvm/kvm-all.c"

echo "--- Patching BPF Headers ---"
# Patch System Header
if ! grep -q "bpf_probe_read_hyperupcall" "$SYS_BPF_HEADER"; then
    echo -e "\n$HELPER_DEF" | sudo tee -a "$SYS_BPF_HEADER"
    echo "[OK] System BPF header patched."
else
    echo "[SKIP] System BPF header already patched."
fi

# Ensure HyperTurtle BPF directory exists and patch header
mkdir -p "$HT_BPF_DIR"
if [ ! -f "$HT_BPF_HEADER" ] || ! grep -q "bpf_probe_read_hyperupcall" "$HT_BPF_HEADER"; then
    echo -e "\n$HELPER_DEF" | sudo tee -a "$HT_BPF_HEADER"
    echo "[OK] HyperTurtle BPF header patched."
else
    echo "[SKIP] HyperTurtle BPF header already patched."
fi

echo "--- Patching QEMU Source ---"
# Python one-liner handles the NULL check safely
python3 -c "
import os
path = '$QEMU_FILE'
if os.path.exists(path):
    with open(path, 'r') as f: content = f.read()
    target = 'if (nc->info->type != NET_CLIENT_DRIVER_TAP)'
    replacement = 'if (nc == NULL) return 0;\n    ' + target
    if target in content and 'nc == NULL' not in content:
        with open(path, 'w') as f: f.write(content.replace(target, replacement))
        print('[OK] QEMU source patched.')
    else:
        print('[SKIP] QEMU already patched or target line missing.')
else:
    print('[ERROR] QEMU file not found!')
"
